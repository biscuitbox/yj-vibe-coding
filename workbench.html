2캔버스 켜기입력창 아래의 캔버스 버튼을 누릅니다.
3프롬프트 붙여넣기아래에서 복사해 그대로 넣습니다.

그다음 — 내 주소 만들기 (1차시에 한 번만)

계정과 저장소는 처음 한 번만 만듭니다. 그 뒤로는 파일을 올리기만 하면 됩니다.

1계정 만들기github.com → Sign up. 아이디가 그대로 주소가 됩니다. 영문 소문자로 짧게.
2저장소 만들기오른쪽 위 + → New repository. 이름은 my-web처럼 영문으로, Public으로.
3인터넷에 공개Settings → Pages → Branch를 main으로 Save.
4주소 확인1~2분 뒤 내아이디.github.io/my-web/이 열립니다.

매 차시 끝 — 내 저장소에 올리기

1차시부터 6차시까지 매번 같은 네 단계입니다. 익숙해지면 1분이면 됩니다.

1코드 받기캔버스 오른쪽 위 메뉴에서 파일을 받습니다.
2이름 바꾸기영문 + .html. 첫 페이지만 index.html.
3업로드내 저장소 → Add file → Upload files → 끌어다 놓기.
4Commit아래 Commit changes. 1~2분 뒤 주소로 열립니다.

같은 이름으로 다시 올리면 덮어쓰기가 됩니다. 고친 파일은 같은 이름으로 올리세요. 올린 직후 404가 뜨면 고장이 아니라 아직 올라가는 중입니다.

    두더지를 누르면 점수가 오르고, 가끔 폭탄이 나오는데 그걸 누르면 점수가 깎여.

[반드시 지킬 규격]
- HTML, CSS, 자바스크립트를 파일 하나에 모두 넣어 줘. 외부 라이브러리는 쓰지 마.
- 한 판은 3분 안에 끝나야 해.
- 결과는 숫자 하나(점수)로 나오고, 높을수록 잘한 거야.
- 코드 맨 위에 const GAME_ID = 'my-game'; 를 만들어 줘.
- 게임이 끝나면 gameOver(점수) 함수를 부르고, 그 함수 안에서 점수를 화면에 보여 줘.
- 혼자 하는 게임이고, 마우스나 손가락 터치만 써.

[디자인]
- 화면 맨 위에 남은 시간과 현재 점수를 크게.
- 글자는 크게, 휴대폰에서도 가로 스크롤 없이 보이게.
- 버튼을 누르면 눌렸다는 게 바로 보이게 해 줘.

[데이터]
- 아직 저장 기능은 넣지 마. 다음 시간에 붙일 거야.

/* =========================================================
   게임 규격 — 이 네 줄은 어떤 게임을 만들어도 똑같이 지킵니다
   ========================================================= */
const GAME_ID   = 'reaction';
const GAME_NAME = '반응속도';
// 게임이 끝나면 gameOver(점수) 를 부릅니다. 점수는 높을수록 잘한 것.

/* ===== 점수 저장 모듈 (3차시) =====
   localStorage = 이 브라우저에만 남는 저장소 */
function saveScore(score){
  const key = 'best_' + GAME_ID;
  const best = Number(localStorage.getItem(key) || 0);
  if (score > best) localStorage.setItem(key, String(score));
  return Math.max(score, best);
}
function loadBest(){
  return Number(localStorage.getItem('best_' + GAME_ID) || 0);
}

/* ===== 게임 본체 ===== */
const TOTAL = 5;
const pad = document.getElementById('pad');
const roundsEl = document.getElementById('rounds');
const scoreEl = document.getElementById('score');
const detailEl = document.getElementById('detail');
const bestEl = document.getElementById('best');
const againBtn = document.getElementById('again');

let times = [], state = 'idle', timer = null, startAt = 0;

function drawRounds(){
  roundsEl.innerHTML = '';
  for (let i = 0; i < TOTAL; i++){
    const d = document.createElement('div');
    d.className = 'dot' + (times[i] !== undefined ? ' on' : '');
    d.textContent = times[i] !== undefined ? times[i] : (i + 1);
    roundsEl.appendChild(d);
  }
}

function setPad(cls, title, desc){
  pad.className = 'pad' + (cls ? ' ' + cls : '');
  pad.innerHTML = '<b>' + title + '</b><span>' + desc + '</span>';
}

function nextRound(){
  if (times.length >= TOTAL){ finish(); return; }
  state = 'wait';
  setPad('wait', '기다리세요', '초록이 될 때까지 누르면 안 됩니다');
  const delay = 1200 + Math.random() * 2600;
  timer = setTimeout(function(){
    state = 'go';
    startAt = performance.now();
    setPad('go', '지금!', '누르세요');
  }, delay);
}

function finish(){
  state = 'done';
  const avg = Math.round(times.reduce(function(a,b){return a+b;}, 0) / times.length);
  const score = Math.max(0, 1000 - avg);
  setPad('', '끝!', '평균 ' + avg + '밀리초');
  detailEl.innerHTML =
    '<div class="row"><span>평균 반응속도</span><b>' + avg + ' ms</b></div>' +
    '<div class="row"><span>가장 빨랐던 판</span><b>' + Math.min.apply(null, times) + ' ms</b></div>';
  gameOver(score);
  againBtn.style.display = 'block';
}

function gameOver(score){
  scoreEl.textContent = score;
  const best = saveScore(score);
  bestEl.textContent = best;
}

pad.addEventListener('click', function(){
  if (state === 'idle' || state === 'done'){
    times = []; drawRounds();
    scoreEl.textContent = '—'; detailEl.innerHTML = '';
    againBtn.style.display = 'none';
    nextRound();
    return;
  }
  if (state === 'wait'){
    clearTimeout(timer);
    state = 'early';
    setPad('early', '너무 빨랐어요', '초록이 될 때까지 기다려야 합니다<br>눌러서 이 판 다시 하기');
    setTimeout(function(){ if (state === 'early') state = 'ready'; }, 400);
    return;
  }
  if (state === 'ready'){ nextRound(); return; }
  if (state === 'go'){
    const ms = Math.round(performance.now() - startAt);
    times.push(ms);
    drawRounds();
    if (times.length >= TOTAL){ finish(); }
    else { setPad('', ms + ' ms', '눌러서 다음 판으로'); state = 'ready'; }
  }
});

againBtn.addEventListener('click', function(){ pad.click(); });

drawRounds();
bestEl.textContent = loadBest();
</script>

  try { const v = JSON.parse(localStorage.getItem(LOCAL_KEY) || '[]'); return Array.isArray(v) ? v : []; }
  catch (e) { return []; }
}
function localSave(){ localStorage.setItem(LOCAL_KEY, JSON.stringify(items)); }

function scheduleSave(){
  el('sync').textContent = '저장 중…';
  clearTimeout(saveTimer);
  saveTimer = setTimeout(doSave, 600);
}

async function doSave(){
  if (cloud && uid){
    try {
      await cloud.setDoc(cloud.doc(cloud.db, 'users', uid, 'data', 'todos'), { items });
      el('sync').textContent = '저장됨';
    } catch (e){
      el('sync').textContent = '저장 실패 — 인터넷을 확인하세요';
      console.error(e);
    }
  } else {
    localSave();
    el('sync').textContent = '저장됨';
  }
}

function render(){
  const list = el('list');
  list.innerHTML = '';
  items.forEach((it) => {
    const li = document.createElement('li');
    if (it.done) li.className = 'done';

    const chk = document.createElement('button');
    chk.className = 'chk';
    chk.textContent = it.done ? '✓' : '';
    chk.setAttribute('aria-label', it.done ? '완료 취소' : '완료로 표시');
    chk.addEventListener('click', () => { it.done = !it.done; render(); scheduleSave(); });

    const txt = document.createElement('span');
    txt.className = 'txt';
    txt.textContent = it.text;

    const del = document.createElement('button');
    del.className = 'del';
    del.textContent = '×';
    del.setAttribute('aria-label', '삭제');
    del.addEventListener('click', () => {
      items = items.filter((x) => x.id !== it.id);
      render(); scheduleSave();
    });

    li.append(chk, txt, del);
    list.appendChild(li);
  });

  const total = items.length;
  const done = items.filter((x) => x.done).length;
  el('empty').style.display = total ? 'none' : 'block';
  el('ratio').textContent = done + ' / ' + total;
  el('fill').style.width = total ? (done / total * 100) + '%' : '0%';
  el('count').textContent = total ? ('남은 일 ' + (total - done) + '개') : '';
  el('cheer').textContent =
    total === 0 ? '할 일을 적어 보세요' :
    done === total ? '오늘 할 일을 다 했어요' :
    done === 0 ? '하나만 먼저 해 볼까요' : '잘하고 있어요';
}

승인된 도메인을 빠뜨리는 사람이 가장 많습니다. Authentication → Settings → 승인된 도메인에 내아이디.github.io를 넣어야 로그인 창이 열립니다. https://를 빼고, 뒤의 저장소 이름도 빼고 도메인만 넣으세요.

캔버스 미리보기에서는 로그인 창이 안 뜰 수 있습니다. 확인은 반드시 올린 뒤 내 주소에서 하세요.

오늘 흐름
① 내 프로젝트 만들기 → ② 로그인·데이터베이스 켜기 → ③ 설정값 복사 → ④ 도메인 등록 → ⑤ 캔버스에서 붙이기 → ⑥ 올리고 내 주소에서 확인 → ⑦ 보안 규칙 게시

안 될 때 화면에 뜨는 글자를 그대로 보세요
auth/unauthorized-domain — 승인된 도메인에 내 주소를 안 넣었습니다.
auth/popup-blocked — 브라우저가 팝업을 막았습니다. 주소창 오른쪽의 팝업 허용을 누르세요.
permission-denied — 보안 규칙이 막은 것입니다. 저장하려는 위치가 users/내uid/ 안인지 확인하세요.
api-key-not-valid — 설정값을 잘못 붙여넣었습니다. 콘솔에서 다시 복사하세요.

마지막에 보안 규칙을 꼭 게시하세요. 테스트 모드로 만든 데이터베이스는 30일 동안 누구나 읽고 쓸 수 있습니다. 아래 규칙을 복사해 Firestore → 규칙 탭에 붙여넣고 게시를 누르세요.
